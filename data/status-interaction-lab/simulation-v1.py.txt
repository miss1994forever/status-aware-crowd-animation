"""Simultaneous visible-expression/body feedback with unchanged navigation chooser."""
from dataclasses import replace,asdict
import time
import numpy as np
from src.predictive_motion_lab import controller as old
from src.crowd_mobility_lab.controller import choose,DEFAULT
from src.expression_layer import personal_library
from .social import CONFIG,PrivateSelf,visible,respond,controlled_library


def simulate(base,spec,mode='C2',config=None,reverse=False,yoke=None):
    cfg={**CONFIG,**(config or {})};states=old.initial(base,spec)
    # Independent of asset ids. These are synthetic private experimental values.
    rng=np.random.default_rng(spec['seed']+987)
    private={s.id:PrivateSelf(float(x)) for s,x in zip(states,rng.choice([-1.,1.],len(states)))}
    personal={s.reference:personal_library(base,s.reference,cfg['personal_strength']) for s in states}
    for s in states:
        lib=controlled_library(personal[s.reference],private[s.id],0.,cfg)
        s.pose=lib.sample(s.clip,s.phase);s.pose_velocity=lib.velocity(s.clip,s.phase,1/lib.periods[s.clip])
    frames=[old.frame(states,base,0.)];decisions=[];histories={s.id:{} for s in states};controls={s.id:0. for s in states}
    durations=[];holds=0;switches=0;arrivals={};edges=set()
    for step in range(round(spec['duration']/.2)):
        start=time.perf_counter();snapshot=states;observed=[visible(s) for s in snapshot];plans={};paths={}
        for s in (snapshot[::-1] if reverse else snapshot):
            if s.arrived:continue
            replay=None if yoke is None else yoke.get((step,s.id),0.)
            r=respond(visible(s),observed,base.names,histories[s.id],controls[s.id],mode,cfg,replay)
            lib=controlled_library(personal[s.reference],private[s.id],r.applied,cfg)
            opt,d=choose(lib,s,observed,spec['walls'],'long_cv',DEFAULT,log=False)
            plans[s.id]=(opt,d,r)
            if opt is not None:
                paths[s.id]=old.rollout(lib,s,opt,np.arange(1,5)*.05)
                controls[s.id]=r.applied
            else:holds+=1
            decisions.append(dict(t=round(step*.2,6),id=s.id,**d,response=asdict(r),executed=opt is not None,control_after=controls[s.id]))
            for o in observed:
                if s.id<o.id and np.linalg.norm(s.pos-o.pos)<1.5:edges.add((s.id,o.id))
        durations.append(time.perf_counter()-start)
        for j in range(4):
            rendered=[]
            for s in snapshot:
                if s.id not in paths:new=replace(s,hold=not s.arrived,speed=0.,pose_velocity=np.zeros_like(s.pose))
                else:
                    p=paths[s.id];clip=int(plans[s.id][0][0,0]);prev=s.pose if j==0 else p['pose'][j-1,0]
                    new=old.State(s.id,p['pos'][j,0],s.goal,float(p['yaw'][j,0]),float(p['travel'][j,0]),float(p['speed'][j,0]),clip,s.reference,float(p['phase'][j,0]),p['pose'][j,0],(p['pose'][j,0]-prev)/.05)
                    if j==3:
                        switches+=int(s.clip!=clip)
                        if np.linalg.norm(new.pos-new.goal)<.22:new.arrived=True;arrivals.setdefault(s.id,(step+1)*.2)
                rendered.append(new)
            frames.append(old.frame(rendered,base,step*.2+(j+1)*.05))
        states=rendered
    return dict(spec=spec,mode=mode,p_weight=0,layer_strength=cfg['personal_strength'],config=cfg,frames=frames,decisions=decisions,
        private_experiment={i:asdict(v) for i,v in private.items()},runtime=dict(planning_seconds=sum(durations),planning_step_ms_p50=float(np.median(durations)*1000),planning_step_ms_p95=float(np.quantile(durations,.95)*1000),agent_count=len(states),switches=switches,emergency_hold_agent_steps=holds,arrival_times=arrivals,interaction_edges=sorted(edges)))
